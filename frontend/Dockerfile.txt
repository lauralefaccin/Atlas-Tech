# --- Estágio 1: Build do React ---
FROM node:20-alpine AS build
WORKDIR /app

# Copia os arquivos de dependências primeiro (otimiza o cache do Docker)
COPY package*.json ./
RUN npm install

# Copia o restante do código e gera o build
COPY . .
RUN npm run build

# --- Estágio 2: Servidor de Produção ---
FROM nginx:stable-alpine

# Copia os arquivos gerados no build para a pasta pública do Nginx
# NOTA: Se você usa Vite, a pasta gerada é 'dist'. Se usa Create React App, é 'build'.
COPY --from=build /app/dist /usr/share/nginx/html

# Copia uma configuração customizada do Nginx se precisar de roteamento (opcional)
# COPY nginx.conf /etc/nginx/conf.d/default.conf

EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]
